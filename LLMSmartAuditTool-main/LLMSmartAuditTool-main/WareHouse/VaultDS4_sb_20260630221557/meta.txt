pragma solidity ^0.8.0;

contract Vault {
    mapping(address => uint256) public balances;
    address public owner;

    constructor() { owner = msg.sender; }

    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }

    // Reentrancy: external call before state update
    function withdraw(uint256 amount) public {
        require(balances[msg.sender] >= amount);
        (bool ok, ) = msg.sender.call{value: amount}("");
        require(ok);
        balances[msg.sender] -= amount;
    }

    // tx.origin authorization bug
    function changeOwner(address newOwner) public {
        require(tx.origin == owner);
        owner = newOwner;
    }

    // Unprotected selfdestruct
    function kill() public {
        selfdestruct(payable(msg.sender));
    }
}

Reentrancy Leading to Potential DoS (withdraw function)
'''
The `withdraw` function performs an external call to `msg.sender.call{value: amount}("")` before updating the user's balance. This reentrancy vulnerability can be exploited by a malicious contract to recursively call `withdraw`, draining the contract's funds and potentially causing a denial of service by exhausting the contract's balance. Additionally, if the external call fails (e.g., due to a malicious fallback function that reverts), the entire transaction reverts, preventing legitimate withdrawals. Mitigation: Apply the checks-effects-interactions pattern by updating the balance before making the external call. Use a reentrancy guard (e.g., OpenZeppelin's ReentrancyGuard) to prevent recursive calls. Also, consider using a pull-over-push pattern to avoid external call failures blocking withdrawals.
'''
